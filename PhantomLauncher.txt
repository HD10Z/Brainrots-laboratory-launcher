local Players = game:GetService("Players")

local isTablet = isMobile and (game:GetService("GuiService"):GetScreenSize().X > 600)

local deviceType = (isMobile or isTablet) and "MOBILE" or "PC"

local mobileScript = "https://raw.githubusercontent.com/HD10Z/Brainrots-laboratory/refs/heads/main/PhantomHubMobile.txt"
local pcScript = "https://raw.githubusercontent.com/HD10Z/Brainrots-laboratory-PC/refs/heads/main/PhantomHubPC.txt"

local scriptToLoad = (deviceType == "MOBILE") and mobileScript or pcScript

task.spawn(function()
    pcall(function()
        loadstring(game:HttpGet(scriptToLoad))()
    end)
end)
